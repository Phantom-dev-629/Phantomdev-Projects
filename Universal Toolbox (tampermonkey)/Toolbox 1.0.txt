// ==UserScript==
// @name         Universal Toolbox
// @author       Phantom-Dev
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Toolbox for any website
// @match        *://*/*
// @grant        none
// ==/UserScript==

(function () {
    'use strict';

    const SETTINGS_KEY = 'toolboxSettings';

    let settings = JSON.parse(localStorage.getItem(SETTINGS_KEY) || '{}');

    settings.theme ??= 'green';
    settings.hotkey ??= '0';
    settings.x ??= null;
    settings.y ??= null;

    function save() {
        localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
    }

    const themes = {
        green: { bg: '#111', border: 'lime', text: 'lime' },
        blue: { bg: '#111', border: '#00aaff', text: '#00aaff' },
        red: { bg: '#111', border: '#ff4444', text: '#ff4444' },
        dark: { bg: '#222', border: '#888', text: '#fff' }
    };

    const gear = document.createElement('button');
    gear.textContent = '🔧';
    gear.style.position = 'fixed';
    gear.style.bottom = '20px';
    gear.style.right = '20px';
    gear.style.width = '50px';
    gear.style.height = '50px';
    gear.style.borderRadius = '50%';
    gear.style.fontSize = '22px';
    gear.style.zIndex = '999999';
    gear.style.cursor = 'pointer';
    document.body.appendChild(gear);

    const panel = document.createElement('div');
    panel.style.position = 'fixed';
    panel.style.bottom = '80px';
    panel.style.right = '20px';
    panel.style.width = '300px';
    panel.style.maxHeight = '500px';
    panel.style.overflowY = 'auto';
    panel.style.padding = '10px';
    panel.style.borderRadius = '10px';
    panel.style.zIndex = '999999';
    panel.style.display = 'none';
    panel.style.fontFamily = 'Arial';

    document.body.appendChild(panel);

    if (settings.x !== null) {
        panel.style.left = settings.x + 'px';
        panel.style.top = settings.y + 'px';
        panel.style.right = 'auto';
        panel.style.bottom = 'auto';
    }

    function applyTheme() {
        const t = themes[settings.theme];
        panel.style.background = t.bg;
        panel.style.color = t.text;
        panel.style.border = `2px solid ${t.border}`;
    }

    applyTheme();

    gear.onclick = () => {
        panel.style.display =
            panel.style.display === 'none' ? 'block' : 'none';
    };

    const title = document.createElement('div');
    title.textContent = '🔧 Toolbox';
    title.style.fontWeight = 'bold';
    title.style.cursor = 'move';
    title.style.marginBottom = '10px';

    panel.appendChild(title);

    let dragging = false;
    let offsetX = 0;
    let offsetY = 0;

    title.addEventListener('mousedown', e => {
        dragging = true;
        offsetX = e.clientX - panel.offsetLeft;
        offsetY = e.clientY - panel.offsetTop;
    });

    document.addEventListener('mousemove', e => {
        if (!dragging) return;

        panel.style.left = e.clientX - offsetX + 'px';
        panel.style.top = e.clientY - offsetY + 'px';
        panel.style.right = 'auto';
        panel.style.bottom = 'auto';
    });

    document.addEventListener('mouseup', () => {
        if (!dragging) return;

        dragging = false;

        settings.x = panel.offsetLeft;
        settings.y = panel.offsetTop;

        save();
    });

    function section(name) {
        const h = document.createElement('h4');
        h.textContent = name;
        h.style.marginBottom = '5px';
        panel.appendChild(h);
    }

    function button(text, fn) {
        const b = document.createElement('button');
        b.textContent = text;
        b.style.margin = '2px';
        b.onclick = fn;
        panel.appendChild(b);
        return b;
    }

    section('🌐 Quick Links');

    button('Google', () => window.open('https://google.com'));
    button('YouTube', () => window.open('https://youtube.com'));
    button('Gmail', () => window.open('https://mail.google.com'));
    button('Copilot', () => window.open('https://copilot.microsoft.com'));
    button('GitHub', () => window.open('https://github.com'));
    button('Discord', () => window.open('https://discord.com'));

    panel.appendChild(document.createElement('hr'));

    section('⚡ Quick Actions');

    button('Reload', () => location.reload());
    button('Back', () => history.back());
    button('Forward', () => history.forward());
    button('Top', () => window.scrollTo(0, 0));
    button('Bottom', () => window.scrollTo(0, document.body.scrollHeight));

    button('Fullscreen', () => {
        if (!document.fullscreenElement)
            document.documentElement.requestFullscreen();
        else
            document.exitFullscreen();
    });

    panel.appendChild(document.createElement('hr'));

    section('🎨 Appearance');

    button('Dark Mode', () => {

    document.documentElement.style.filter =
        'invert(1) hue-rotate(180deg)';

    document.querySelectorAll('img, video').forEach(el => {
        el.style.filter =
            'invert(1) hue-rotate(180deg)';
    });

});


    button('Normal Mode', () => {
        document.documentElement.style.filter = '';
    });

    button('Hide Images', () => {
        document.querySelectorAll('img').forEach(img => {
            img.style.display = 'none';
        });
    });

    button('Show Images', () => {
        document.querySelectorAll('img').forEach(img => {
            img.style.display = '';
        });
    });

    button('Text +', () => {
        document.body.style.fontSize = 'larger';
    });

    button('Text -', () => {
        document.body.style.fontSize = 'smaller';
    });

    panel.appendChild(document.createElement('hr'));

    section('🔍 Search Selected Text');

    function getSelected() {
        return encodeURIComponent(window.getSelection().toString());
    }

    button('Google Search', () => {
        const t = getSelected();
        if (t) window.open(`https://www.google.com/search?q=${t}`);
    });

    button('YouTube Search', () => {
        const t = getSelected();
        if (t) window.open(`https://www.youtube.com/results?search_query=${t}`);
    });

    button('Wikipedia Search', () => {
        const t = getSelected();
        if (t) window.open(`https://en.wikipedia.org/wiki/Special:Search?search=${t}`);
    });

    panel.appendChild(document.createElement('hr'));

    section('⚙ Settings');

    const themeSelect = document.createElement('select');

    ['green','blue','red','dark'].forEach(theme => {
        const option = document.createElement('option');
        option.value = theme;
        option.textContent = theme;
        themeSelect.appendChild(option);
    });

    themeSelect.value = settings.theme;

    themeSelect.onchange = () => {
        settings.theme = themeSelect.value;
        save();
        applyTheme();
    };

    panel.appendChild(themeSelect);

    panel.appendChild(document.createElement('br'));
    panel.appendChild(document.createElement('br'));

    const hotkeyInput = document.createElement('input');
    hotkeyInput.value = settings.hotkey;

    const hotkeyBtn = document.createElement('button');
    hotkeyBtn.textContent = 'Save Hotkey';

    hotkeyBtn.onclick = () => {
        settings.hotkey = hotkeyInput.value;
        save();
        alert('Hotkey saved');
    };

    panel.appendChild(hotkeyInput);
    panel.appendChild(hotkeyBtn);

document.addEventListener('keydown', e => {

    if (
        e.target.tagName === 'INPUT' ||
        e.target.tagName === 'TEXTAREA'
    ) return;

    if (e.code === 'Digit0') {
        panel.style.display =
            panel.style.display === 'none'
                ? 'block'
                : 'none';
    }
});

})();
